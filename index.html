-- BLUESTAR PRIME ⭐ TRADING JOURNAL -- Version 1 database create table public.traders ( telegram_id bigint primary key, first_name text, last_name text, username text, created_at timestamptz not null default now(), updated_at timestamptz not null default now() ); create table public.trades ( id bigint generated by default as identity primary key, telegram_id bigint not null references public.traders(telegram_id) on delete cascade, pair text not null, trading_date date not null, direction text not null check (direction in ('Long', 'Short')), entry_price numeric, exit_price numeric, stop_loss numeric, take_profit numeric, trade_amount numeric, risk_percent numeric, result text not null check (result in ('Win', 'Loss', 'Breakeven')), pnl_amount numeric not null default 0, notes text, created_at timestamptz not null default now() ); -- Enable Row Level Security alter table public.traders enable row level security; alter table public.trades enable row level security; -- Users will only be able to access their own records. -- The secure Telegram authentication/backend will provide -- telegram_id in the user's authenticated JWT. create policy "traders can view their own profile" on public.traders for select using ( telegram_id = (auth.jwt() ->> 'telegram_id')::bigint ); create policy "traders can insert their own profile" on public.traders for insert with check ( telegram_id = (auth.jwt() ->> 'telegram_id')::bigint ); create policy "traders can update their own profile" on public.traders for update using ( telegram_id = (auth.jwt() ->> 'telegram_id')::bigint ) with check ( telegram_id = (auth.jwt() ->> 'telegram_id')::bigint ); create policy "traders can view their own trades" on public.trades for select using ( telegram_id = (auth.jwt() ->> 'telegram_id')::bigint ); create policy "traders can insert their own trades" on public.trades for insert with check ( telegram_id = (auth.jwt() ->> 'telegram_id')::bigint ); create policy "traders can update their own trades" on public.trades for update using ( telegram_id = (auth.jwt() ->> 'telegram_id')::bigint ) with check ( telegram_id = (auth.jwt() ->> 'telegram_id')::bigint ); create policy "traders can delete their own trades" on public.trades for delete using ( telegram_id = (auth.jwt() ->> 'telegram_id')::bigint );
